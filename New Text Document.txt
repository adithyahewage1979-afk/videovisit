    const videos = [
      {
        id: 1,

      },
      {
        id: 3,

        description: "ඔයා වෙනුවෙන් තෝරාගත් වීඩියෝව.",

      },
      {
        id: 4,

        thumb: ""
      },
      {
        id: 5,
        title: "Premium Video 05",
        description: "තවත් වටිනා වීඩියෝවක්.",
        price: 850,
        category: "Premium",
        thumb: ""
      },
      {
        id: 6,
        title: "Special Video 06",
        description: "නවතම එකතුවෙන්.",
        price: 1200,
        category: "Special",
        thumb: ""
      }
    ];

    const grid = document.getElementById("videoGrid");
    const search = document.getElementById("search");
    const sort = document.getElementById("sort");

    function money(price) {
      return "Rs. " + Number(price).toLocaleString("en-LK");
    }

    function makeWhatsAppLink(message) {
      return "https://wa.me/" + WHATSAPP_NUMBER +
        "?text=" + encodeURIComponent(message);
    }

    function renderVideos() {
      const query = search.value.toLowerCase().trim();
      const sortOrder = sort.value;

      let filtered = videos.filter(video => {
        const text = (
          video.title + " " +
          video.description + " " +
          video.category
        ).toLowerCase();

        return text.includes(query);
      });

      if (sortOrder === "low") {
        filtered.sort((a, b) => a.price - b.price);
      } else if (sortOrder === "high") {
        filtered.sort((a, b) => b.price - a.price);
      }

      grid.replaceChildren();

      if (filtered.length === 0) {
        const empty = document.createElement("p");
        empty.className = "empty";
        empty.textContent = "වීඩියෝ හමු වුණේ නැහැ.";
        grid.appendChild(empty);
        return;
      }

      filtered.forEach(video => {
        const card = document.createElement("article");
        card.className = "card";

        const thumb = document.createElement("div");
        thumb.className = "thumb";

        const play = document.createElement("span");
        play.className = "play";
        play.textContent = "▶";
        thumb.appendChild(play);

        if (video.thumb) {
          const img = document.createElement("img");
          img.src = video.thumb;
          img.alt = video.title;
          img.loading = "lazy";

          img.onerror = () => img.remove();
          thumb.appendChild(img);
        }

        const price = document.createElement("span");
        price.className = "price";
        price.textContent = money(video.price);
        thumb.appendChild(price);

        const info = document.createElement("div");
        info.className = "info";

        const category = document.createElement("span");
        category.className = "category";
        category.textContent = video.category;

        const title = document.createElement("h3");
        title.textContent = video.title;

        const desc = document.createElement("p");
        desc.textContent = video.description;

        const buy = document.createElement("a");
        buy.className = "primary buy";
        buy.textContent = "Buy Now";
        buy.href = makeWhatsAppLink(
          "Hello! I want to purchase a video.\n" +
          "Video: " + video.title + "\n" +
          "Video ID: " + video.id + "\n" +
          "Price: " + money(video.price) + "\n" +
          "Please send me the payment instructions."
        );
        buy.target = "_blank";
        buy.rel = "noopener";

        info.append(category, title, desc, buy);
        card.append(thumb, info);
        grid.appendChild(card);
      });
    }

    search.addEventListener("input", renderVideos);
    sort.addEventListener("change", renderVideos);

    document.getElementById("contactLink").href =
      makeWhatsAppLink(
        "Hello! I would like to know more about your videos."
      );

    document.getElementById("year").textContent =
      new Date().getFullYear();

    renderVideos();
  </script>
